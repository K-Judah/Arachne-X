set(calibration_sources src/record.cpp src/repository.cpp)
if(ESP_PLATFORM)
    idf_component_register(SRCS ${calibration_sources} INCLUDE_DIRS include REQUIRES control_core)
    target_compile_features(${COMPONENT_LIB} PUBLIC cxx_std_17)
    target_compile_options(${COMPONENT_LIB} PRIVATE -Wall -Wextra -Wpedantic -Werror)
else()
    foreach(mode IN ITEMS simulation locked)
        add_library(calibration_${mode} STATIC ${calibration_sources})
        target_include_directories(calibration_${mode} PUBLIC include)
        target_link_libraries(calibration_${mode} PUBLIC arachne_${mode})
        if(MSVC)
            target_compile_options(calibration_${mode} PRIVATE /W4 /WX /permissive-)
        else()
            target_compile_options(calibration_${mode} PRIVATE -Wall -Wextra -Wpedantic -Werror)
        endif()
    endforeach()
endif()
