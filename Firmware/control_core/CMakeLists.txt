set(core_sources src/configuration.cpp src/actuator.cpp)
if(ESP_PLATFORM)
    idf_component_register(SRCS ${core_sources} INCLUDE_DIRS include)
    target_compile_features(${COMPONENT_LIB} PUBLIC cxx_std_17)
    target_compile_options(${COMPONENT_LIB} PRIVATE -Wall -Wextra -Wpedantic -Werror)
else()
    # Compile the exact same sources twice: simulation enabled and firmware locked.
    foreach(target IN ITEMS arachne_simulation arachne_locked)
        add_library(${target} STATIC ${core_sources})
        target_include_directories(${target} PUBLIC include)
        target_compile_features(${target} PUBLIC cxx_std_17)
        if(MSVC)
            target_compile_options(${target} PRIVATE /W4 /WX /permissive-)
        else()
            target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
        endif()
    endforeach()
    target_compile_definitions(arachne_simulation PRIVATE ARACHNE_HOST_SIMULATION=1)
endif()
