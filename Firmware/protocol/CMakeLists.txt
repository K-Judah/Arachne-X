set(protocol_sources src/wire.cpp src/endpoint.cpp)
if(ESP_PLATFORM)
    idf_component_register(SRCS ${protocol_sources} INCLUDE_DIRS include REQUIRES control_core)
    target_compile_features(${COMPONENT_LIB} PUBLIC cxx_std_17)
    target_compile_options(${COMPONENT_LIB} PRIVATE -Wall -Wextra -Wpedantic -Werror)
else()
    foreach(mode IN ITEMS simulation locked)
        add_library(protocol_${mode} STATIC ${protocol_sources})
        target_include_directories(protocol_${mode} PUBLIC include)
        target_link_libraries(protocol_${mode} PUBLIC arachne_${mode})
        if(MSVC)
            target_compile_options(protocol_${mode} PRIVATE /W4 /WX /permissive-)
        else()
            target_compile_options(protocol_${mode} PRIVATE -Wall -Wextra -Wpedantic -Werror)
        endif()
    endforeach()
endif()
